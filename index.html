{

Assignment2/linker/exp2.c 
#include <p18f4550.h>
#include "vector_relocate.h"

void delay(unsigned int time)
{
    unsigned int i, j;
    for (i = 0; i < time; i++)
        for (j = 0; j < 710; j++);
}

void main(void)
{
    ADCON1 = 0x0F;
    INTCON2bits.RBPU = 0;
    TRISBbits.TRISB0 = 1;
    TRISBbits.TRISB1 = 1;
    TRISCbits.TRISC1 = 0;
    TRISCbits.TRISC2 = 0;
    TRISD = 0x00;

    PORTD = 0x00;
    LATCbits.LATC1 = 0;
    LATCbits.LATC2 = 0;

    while (1)
    {
        if (PORTBbits.RB1 == 0) /* SW0 pressed */
        {
            LATCbits.LATC1 = 1; /* Relay ON */
            LATCbits.LATC2 = 1; /* Buzzer ON */
            PORTD = 0x01;
            while (PORTD != 0x80)
            {
                delay(250);
                PORTD = PORTD << 1;
            }
            delay(250);
        }

        if (PORTBbits.RB0 == 0) /* SW1 pressed */
        {
            LATCbits.LATC1 = 0; /* Relay OFF */
            LATCbits.LATC2 = 0; /* Buzzer OFF */
            PORTD = 0x80;
            while (PORTD != 0x01)
            {
                delay(250);
                PORTD = PORTD >> 1;
            }
            delay(250);
        }
    }
}

Expt 3: Timer 0 Interrupt 10 Hz Square Wave Generation

PIC18F4550 • 10 Hz

Mathematical Derivation (10 Hz Wave)

Target Frequency ($F$)10 Hz (Total Period T = 100 ms)
Half-Cycle Delay ($T_d$)T / 2 = 50 ms = 50,000 μs
Oscillator Frequency ($F_{osc}$)48 MHz → Fcy = 12 MHz
Instruction Period ($T_p$)4 / 48 MHz = 1/12 μs
Prescaler (T0CON = 0x03)1:16 → Teq = 16/12 = 4/3 μs
Required Counts ($n$)50,000 / (4/3) = 37,500 counts
Initial Register Value65,536 - 37,500 = 28,036 = 0x6D84
Reload RegistersTMR0H = 0x6D, TMR0L = 0x84
Interrupt Vector0x1008 (High Priority)
Assignment3/exp3.c 
#include <p18f4550.h>

#define TMR0_RELOAD_H 0x6D
#define TMR0_RELOAD_L 0x84

void timer0_isr(void);
extern void _startup(void);

#pragma code _RESET_INTERRUPT_VECTOR = 0x1000
void _reset(void) { _asm goto _startup _endasm }
#pragma code

#pragma code _HIGH_INTERRUPT_VECTOR = 0x1008
void _high_ISR(void) { _asm goto timer0_isr _endasm }
#pragma code

#pragma interrupt timer0_isr
void timer0_isr(void)
{
    if (INTCONbits.TMR0IF)
    {
        PORTBbits.RB0 = ~PORTBbits.RB0;
        INTCONbits.TMR0IF = 0;
        TMR0H = TMR0_RELOAD_H;
        TMR0L = TMR0_RELOAD_L;
    }
}

void main(void)
{
    ADCON1 = 0x0F;
    TRISBbits.TRISB0 = 0;
    PORTBbits.RB0 = 0;

    T0CON = 0x03; /* 16-bit, 1:16 Prescaler */
    TMR0H = TMR0_RELOAD_H;
    TMR0L = TMR0_RELOAD_L;

    INTCONbits.TMR0IF = 0;
    INTCONbits.TMR0IE = 1;
    INTCONbits.GIE = 1;
    T0CONbits.TMR0ON = 1;

    while (1);
}

Assignment4/exp4.c 
#include <p18f4550.h>

void set_pwm_duty_10bit(unsigned int duty_10bit)
{
    CCPR1L = (unsigned char)(duty_10bit >> 2);
    CCP1CONbits.DC1B1 = (duty_10bit >> 1) & 0x01;
    CCP1CONbits.DC1B0 = duty_10bit & 0x01;
}

void main(void)
{
    unsigned int steps[5] = {150, 300, 450, 600, 750};
    int i;

    ADCON1 = 0x0F;
    TRISBbits.TRISB0 = 0;
    TRISBbits.TRISB1 = 0;
    TRISCbits.TRISC2 = 0;

    PORTBbits.RB0 = 1; /* Forward */
    PORTBbits.RB1 = 0;

    PR2 = 186;       /* 4 kHz at 48 MHz */
    CCP1CON = 0x0C;   /* PWM Mode */
    T2CON = 0x06;     /* Prescaler 1:16, Timer 2 ON */

    while (1)
    {
        for (i = 0; i < 5; i++)
        {
            set_pwm_duty_10bit(steps[i]);
            delay(1500);
        }
        for (i = 4; i >= 0; i--)
        {
            set_pwm_duty_10bit(steps[i]);
            delay(1500);
        }
    }
}

Assignment5/exp5.c 
#include <p18f4550.h>

#define RS PORTEbits.RE0
#define RW PORTEbits.RE1
#define EN PORTEbits.RE2

void Lcd_cmd(unsigned char cmd)
{
    PORTD = cmd;
    RS = 0; RW = 0; EN = 1;
    delay(5); EN = 0; delay(5);
}

void Lcd_data(unsigned char data)
{
    PORTD = data;
    RS = 1; RW = 0; EN = 1;
    delay(5); EN = 0; delay(5);
}

void Lcd_init(void)
{
    delay(15);
    Lcd_cmd(0x38);
    Lcd_cmd(0x01);
    Lcd_cmd(0x0E);
    Lcd_cmd(0x06);
    Lcd_cmd(0x80);
}

void main(void)
{
    ADCON1 = 0x0F;
    TRISD = 0x00;
    TRISE = 0x00;
    Lcd_init();
    Lcd_string("HELLO WORLD");
    while (1);
}

#include <WiFi.h>

const char* ssid = "YOUR_SSID";
const char* password = "YOUR_PASSWORD";

void setup()
{
    Serial.begin(115200);
    pinMode(2, OUTPUT);
    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED)
    {
        digitalWrite(2, !digitalRead(2));
        delay(300);
    }
    digitalWrite(2, HIGH);
    Serial.println("Connected! IP: " + WiFi.localIP().toString());
}

/* STM32 TIM2 Millisecond Hardware Timer */
void TIM2_Init(void)
{
    RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
    TIM2->PSC = 72 - 1; /* 1 MHz count clock */
    TIM2->ARR = 1000 - 1; /* 1 ms overflow period */
    TIM2->CR1 |= TIM_CR1_CEN;
}
